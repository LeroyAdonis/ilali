import { betterAuth } from "better-auth";
import { drizzleAdapter } from "better-auth/adapters/drizzle";
import { db } from "@/lib/db/index";
import * as schema from "@/lib/db/schema";

export const auth = betterAuth({
  database: drizzleAdapter(db, {
    provider: "pg",
    schema: {
      user: schema.users,
    },
  }),
  emailAndPassword: {
    enabled: true,
  },
  // Optional: session config
  session: {
    expiresIn: 30 * 24 * 60 * 60, // 30 days
  },
});

## T033: Create `src/middleware.ts` — Admin Route Protection

Next.js middleware that protects `/admin/*` routes:
- Check for auth session cookie
- If no session → redirect to `/auth/signin`
- If session exists but user role is not "admin" → redirect to `/` 
- Allow public routes through

```typescript
import { NextRequest, NextResponse } from "next/server";
import { getSessionCookie } from "better-auth";

export async function middleware(request: NextRequest) {
  const { pathname } = request.nextUrl;
  
  if (pathname.startsWith("/admin")) {
    const session = await auth.api.getSession({ headers: request.headers });
    if (!session) {
      return NextResponse.redirect(new URL("/auth/signin", request.url));
    }
    // Optionally check admin role
  }
  return NextResponse.next();
}

export const config = {
  matcher: ["/admin/:path*"],
};
```

NOTE: Check better-auth 1.6.25 for the exact middleware/session API. The patterns above may need adjustment for the installed version.
